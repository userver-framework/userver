project(userver-chaotic-openapi-integration-tests CXX)

file(GLOB_RECURSE SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*pp)
add_executable(${PROJECT_NAME} ${SOURCES})
add_google_tests(${PROJECT_NAME})

target_link_libraries(${PROJECT_NAME} userver-chaotic-openapi userver-universal-utest userver-utest)

file(
    GLOB CLIENTS
    RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/clients/
    ${CMAKE_CURRENT_SOURCE_DIR}/clients/*
)
foreach(CLIENT ${CLIENTS})
    string(REPLACE "-" "_" CLIENT_NAME ${CLIENT})
    file(GLOB SCHEMAS ${CMAKE_CURRENT_SOURCE_DIR}/clients/${CLIENT}/*.yaml)
    userver_target_generate_openapi_client(
        ${PROJECT_NAME}-client-${CLIENT}
        NAME ${CLIENT_NAME}
        SCHEMAS ${SCHEMAS}
    )
    target_link_libraries(${PROJECT_NAME} ${PROJECT_NAME}-client-${CLIENT})
endforeach()

file(
    GLOB HANDLERS
    RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/handlers/
    ${CMAKE_CURRENT_SOURCE_DIR}/handlers/*
)
foreach(HANDLER ${HANDLERS})
    file(GLOB SCHEMAS ${CMAKE_CURRENT_SOURCE_DIR}/handlers/${HANDLER}/*.yaml)
    userver_target_generate_openapi_handlers(
        ${PROJECT_NAME}-handler-${HANDLER}
        NAME ${HANDLER}
        SCHEMAS ${SCHEMAS}
        SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src
    )
    target_link_libraries(${PROJECT_NAME} ${PROJECT_NAME}-handler-${HANDLER})
endforeach()

# Documents with x-usrv-cpp-namespace: every document is generated once, by the target it is an input of,
# and included by the targets referring to it. The types of an operations document are generated by a target the
# first of its client and handlers adds.
set(TAGGED_DIR ${CMAKE_CURRENT_SOURCE_DIR}/tagged)
userver_target_generate_chaotic(
    ${PROJECT_NAME}-tagged-schemas SCHEMAS ${TAGGED_DIR}/common/errors.yaml ${TAGGED_DIR}/schemas/credentials.yaml
)
userver_target_generate_openapi_client(
    ${PROJECT_NAME}-tagged-client
    NAME tagged_api
    SCHEMAS ${TAGGED_DIR}/api/vault.yaml
)
target_link_libraries(${PROJECT_NAME}-tagged-client ${PROJECT_NAME}-tagged-schemas)
target_link_libraries(${PROJECT_NAME} ${PROJECT_NAME}-tagged-client)
foreach(TAGGED_HANDLER api admin)
    userver_target_generate_openapi_handlers(
        ${PROJECT_NAME}-tagged-handler-${TAGGED_HANDLER}
        NAME tagged_${TAGGED_HANDLER}
        SCHEMAS ${TAGGED_DIR}/${TAGGED_HANDLER}/vault.yaml
        SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src
    )
    target_link_libraries(${PROJECT_NAME}-tagged-handler-${TAGGED_HANDLER} ${PROJECT_NAME}-tagged-schemas)
    target_link_libraries(${PROJECT_NAME} ${PROJECT_NAME}-tagged-handler-${TAGGED_HANDLER})
endforeach()
